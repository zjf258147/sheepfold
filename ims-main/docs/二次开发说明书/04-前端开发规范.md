## 4.3 API 封装规范

### 4.3.1 文件组织

每个业务模块一个文件，放在 `src/api/` 下：

```javascript
// src/api/my_module.js
import request from '@/utils/request'

// 列表查询
export function listItems(params) {
  return request.get('/api/v1/my-module/items', { params })
}

// 详情
export function getItem(id) {
  return request.get(`/api/v1/my-module/items/${id}`)
}

// 新增
export function createItem(data) {
  return request.post('/api/v1/my-module/items', data)
}

// 更新
export function updateItem(id, data) {
  return request.put(`/api/v1/my-module/items/${id}`, data)
}

// 删除
export function deleteItem(id) {
  return request.delete(`/api/v1/my-module/items/${id}`)
}

// 文件下载
export function exportItems(params) {
  return request.get('/api/v1/my-module/items/export', {
    params,
    responseType: 'blob',
  })
}
```

### 4.3.2 请求拦截器

[frontend/src/utils/request.js](file:///c:/Users/25075/Desktop/IMS生产物料与产品追溯管理系统/ims-main/frontend/src/utils/request.js)

```javascript
const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '',
  timeout: 30000,
})

// 请求拦截器：自动注入 Token
request.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器：自动处理错误
request.interceptors.response.use(
  (response) => {
    const res = response.data
    if (res.code !== undefined && res.code !== 0) {
      ElMessage.error(res.msg || '请求失败')
      return Promise.reject(new Error(res.msg))
    }
    return res  // ← 直接返回 data，调用方拿到的就是 { code, msg, data }
  },
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token')
      router.push('/login')
      ElMessage.error('登录已过期，请重新登录')
    } else {
      ElMessage.error(error.response?.data?.detail || '网络错误')
    }
    return Promise.reject(error)
  }
)
```

> ⚠️ 响应拦截器已处理 `R.fail()` 的 `code !== 0` 情况，并用 `ElMessage.error` 提示。在页面中调用 API 时无需额外判断 `code`。

### 4.3.3 在页面中调用 API

```javascript
import { listItems, createItem, deleteItem } from '@/api/my_module'

const loading = ref(false)
const list = ref([])

async function fetchData() {
  loading.value = true
  try {
    const res = await listItems({ page: page.value, page_size: pageSize.value, ...query.value })
    list.value = res.data.items
    total.value = res.data.total
  } catch {
    // 错误已由拦截器统一提示
  } finally {
    loading.value = false
  }
}

async function handleSubmit() {
  submitLoading.value = true
  try {
    await createItem(form.value)
    ElMessage.success('新增成功')
    dialogVisible.value = false
    fetchData()
  } catch {
    // 错误已由拦截器统一提示
  } finally {
    submitLoading.value = false
  }
}
```

## 4.4 状态管理规范（Pinia）

### 4.4.1 Store 定义

使用 `defineStore` + Composition API：

```javascript
// src/stores/my_store.js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useMyStore = defineStore('myStore', () => {
  // 状态
  const data = ref(null)

  // 计算属性
  const isEmpty = computed(() => !data.value)

  // 操作
  async function fetch() {
    const res = await apiCall()
    data.value = res.data
  }

  function reset() {
    data.value = null
  }

  return { data, isEmpty, fetch, reset }
})
```

### 4.4.2 现有 Store

| Store | 文件 | 职责 |
|-------|------|------|
| `useAuthStore` | `stores/auth.js` | 用户信息、Token、角色判断 |
| `useBrandStore` | `stores/brand.js` | 品牌配置（名称、Logo、副标题） |

### 4.4.3 在组件中使用

```javascript
import { useAuthStore } from '@/stores/auth'

const auth = useAuthStore()

// 读取
console.log(auth.user.username)
console.log(auth.isAdmin)

// 调用
await auth.login(username, password)
await auth.fetchUser()
```

> 💡 只将需要跨组件共享的状态放入 Store。页面内部状态用 `ref`/`reactive` 即可。

## 4.5 路由配置规范

### 4.5.1 路由定义

文件：[frontend/src/router/index.js](file:///c:/Users/25075/Desktop/IMS生产物料与产品追溯管理系统/ims-main/frontend/src/router/index.js)

```javascript
{
  path: '/my-module',
  name: 'MyModule',
  component: () => import('@/views/MyModule.vue'),
  meta: {
    title: '我的模块',          // 页面标题
    roles: ['ADMIN', 'WAREHOUSE'],  // 允许访问的角色（ADMIN 自动放行）
    // admin: true,              // 仅管理员
  },
}
```

### 4.5.2 路由守卫逻辑

1. 未登录访问 → 跳转 `/login`
2. 已登录访问 `/login` → 跳转 `/dashboard`
3. 页面有 `meta.roles` → 检查用户角色，无权限跳转 `/dashboard`
4. 页面有 `meta.admin` → 检查 `auth.isAdmin`

### 4.5.3 新增路由步骤

1. 在 `routes` 数组的 `children` 中添加路由对象
2. 若页面在菜单显示，同步更新 `MainLayout.vue` 的 `allMenus` 数组

> ⚠️ 菜单配置和路由配置是两套独立的结构，新增页面时两者都需更新。

**属性说明**：

### 4.7.3 菜单权限控制

```javascript
const menus = computed(() => {
  const role = auth.role || ''
  return allMenus.filter((m) => {
    if (m.adminOnly && !auth.isAdmin) return false
    if (m.roles && !m.roles.includes(role) && !auth.isAdmin) return false
    return true
  })
})
```

> 新增页面时，`allMenus` 和 `routes` 的角色配置必须一致。

## 4.10 新增页面完整示例

以"FAQ管理"为例。

### Step 1: 创建 API 封装 `src/api/faq.js`

```javascript
import request from '@/utils/request'

export function listFaq(params) {
  return request.get('/api/v1/faq', { params })
}

export function createFaq(data) {
  return request.post('/api/v1/faq', data)
}

export function updateFaq(id, data) {
  return request.put(`/api/v1/faq/${id}`, data)
}

export function deleteFaq(id) {
  return request.delete(`/api/v1/faq/${id}`)
}
```

### Step 2: 创建页面 `src/views/Faq.vue`

```vue
<script setup>
import { onMounted, ref } from 'vue'
import { Delete, Edit } from '@element-plus/icons-vue'
import { listFaq, createFaq, updateFaq, deleteFaq } from '@/api/faq'

const loading = ref(false)
const list = ref([])
const total = ref(0)
const page = ref(1)
const pageSize = ref(15)

const dialogVisible = ref(false)
const editingId = ref(null)
const submitLoading = ref(false)
const formRef = ref()
const form = ref({ question: '', answer: '', sort_order: 0, status: 'PUBLISHED' })

const rules = {
  question: [{ required: true, message: '请输入问题', trigger: 'blur' }],
  answer: [{ required: true, message: '请输入答案', trigger: 'blur' }],
}

async function fetchData() {
  loading.value = true
  try {
    const res = await listFaq({ page: page.value, page_size: pageSize.value })
    list.value = res.data.items
    total.value = res.data.total
  } finally {
    loading.value = false
  }
}

function handleAdd() {
  editingId.value = null
  form.value = { question: '', answer: '', sort_order: 0, status: 'PUBLISHED' }
  dialogVisible.value = true
}

function handleEdit(row) {
  editingId.value = row.id
  form.value = { ...row }
  dialogVisible.value = true
}

async function handleSubmit() {
  const valid = await formRef.value.validate().catch(() => false)
  if (!valid) return
  submitLoading.value = true
  try {
    if (editingId.value) {
      await updateFaq(editingId.value, form.value)
    } else {
      await createFaq(form.value)
    }
    ElMessage.success(editingId.value ? '更新成功' : '新增成功')
    dialogVisible.value = false
    fetchData()
  } finally {
    submitLoading.value = false
  }
}

async function handleDelete(row) {
  try {
    await ElMessageBox.confirm(`确定删除"${row.question}"吗？`, '确认删除', { type: 'warning' })
    await deleteFaq(row.id)
    ElMessage.success('删除成功')
    fetchData()
  } catch { /* 取消删除 */ }
}

onMounted(fetchData)
</script>

<template>
  <div class="faq-page">
    <div class="toolbar">
      <el-button type="primary" :icon="Plus" @click="handleAdd">新增FAQ</el-button>
    </div>

    <el-table :data="list" v-loading="loading" border stripe>
      <el-table-column prop="id" label="ID" width="60" />
      <el-table-column prop="question" label="问题" min-width="200" />
      <el-table-column prop="sort_order" label="排序" width="80" align="center" />
      <el-table-column prop="status" label="状态" width="80" />
      <el-table-column label="操作" width="150" fixed="right">
        <template #default="{ row }">
          <el-button type="primary" link :icon="Edit" @click="handleEdit(row)">编辑</el-button>
          <el-button type="danger" link :icon="Delete" @click="handleDelete(row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <el-pagination
      v-model:current-page="page" v-model:page-size="pageSize"
      :total="total" :page-sizes="[10, 15, 20, 50]"
      layout="total, sizes, prev, pager, next, jumper" @change="fetchData"
    />

    <el-dialog v-model="dialogVisible" :title="editingId ? '编辑FAQ' : '新增FAQ'" width="600px">
      <el-form :model="form" :rules="rules" ref="formRef" label-width="80px">
        <el-form-item label="问题" prop="question">
          <el-input v-model="form.question" maxlength="500" />
        </el-form-item>
        <el-form-item label="答案" prop="answer">
          <el-input v-model="form.answer" type="textarea" :rows="4" />
        </el-form-item>
        <el-form-item label="排序">
          <el-input-number v-model="form.sort_order" :min="0" />
        </el-form-item>
        <el-form-item label="状态">
          <el-select v-model="form.status">
            <el-option value="PUBLISHED" label="已发布" />
            <el-option value="DRAFT" label="草稿" />
          </el-select>
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" @click="handleSubmit" :loading="submitLoading">确定</el-button>
      </template>
    </el-dialog>
  </div>
</template>
```
