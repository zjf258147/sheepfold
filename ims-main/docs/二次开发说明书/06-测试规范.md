### 6.2.3 API 封装测试模板

```javascript
// frontend/src/__tests__/api/faq.test.js
import { describe, it, expect, beforeEach, vi } from 'vitest'

vi.mock('@/utils/request', () => {
  const mockRequest = {
    get: vi.fn(),
    post: vi.fn(),
    put: vi.fn(),
    delete: vi.fn(),
  }
  return { default: mockRequest }
})

import request from '@/utils/request'
import { listFaq, createFaq, updateFaq, deleteFaq } from '@/api/faq'

describe('FAQ API', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('listFaq 发送 GET 请求并带分页参数', async () => {
    request.get.mockResolvedValue({ code: 0, data: { items: [], total: 0 } })

    const res = await listFaq({ page: 1, page_size: 20 })
    expect(request.get).toHaveBeenCalledWith('/api/v1/faq', {
      params: { page: 1, page_size: 20 },
    })
    expect(res.data.items).toEqual([])
  })

  it('createFaq 发送 POST 请求', async () => {
    request.post.mockResolvedValue({ code: 0, data: { id: 1 } })

    await createFaq({ question: 'Q', answer: 'A' })
    expect(request.post).toHaveBeenCalledWith('/api/v1/faq', {
      question: 'Q', answer: 'A',
    })
  })
})
```

### 6.3.3 测试用例模板

```python
# backend/tests/test_faq.py
import pytest
from httpx import AsyncClient
from app.main import app

@pytest.mark.anyio
async def test_list_faq():
    """测试 FAQ 列表接口。"""
    async with AsyncClient(app=app, base_url="http://test") as client:
        # 先登录获取 Token
        login_res = await client.post("/api/v1/auth/login", json={
            "username": "admin",
            "password": "admin123",
        })
        token = login_res.json()["data"]["access_token"]
        headers = {"Authorization": f"Bearer {token}"}

        # 请求 FAQ 列表
        res = await client.get("/api/v1/faq", headers=headers, params={
            "page": 1,
            "page_size": 10,
        })
        assert res.status_code == 200
        data = res.json()
        assert data["code"] == 0
        assert "items" in data["data"]


@pytest.mark.anyio
async def test_create_faq():
    """测试创建 FAQ。"""
    async with AsyncClient(app=app, base_url="http://test") as client:
        login_res = await client.post("/api/v1/auth/login", json={
            "username": "admin", "password": "admin123",
        })
        token = login_res.json()["data"]["access_token"]
        headers = {"Authorization": f"Bearer {token}"}

        res = await client.post("/api/v1/faq", headers=headers, json={
            "question": "测试问题",
            "answer": "测试答案",
        })
        assert res.status_code == 200
        assert res.json()["code"] == 0


@pytest.mark.anyio
async def test_create_faq_unauthorized():
    """测试未登录时创建 FAQ 返回 401。"""
    async with AsyncClient(app=app, base_url="http://test") as client:
        res = await client.post("/api/v1/faq", json={
            "question": "测试",
            "answer": "测试",
        })
        assert res.status_code == 401
```

## 6.4 测试清单（新增功能必测）

### 6.4.1 API 层

| 测试项 | 必须 | 说明 |
|--------|:---:|------|
| 正常列表查询 | ✅ | 分页参数正确 |
| 空列表查询 | ✅ | 无数据时返回空数组 |
| 正常创建 | ✅ | 返回新记录的完整数据 |
| 必填字段缺失 | ✅ | 返回 422 验证错误 |
| 未登录调用 | ✅ | 返回 401 |
| 无权限调用 | ✅ | 返回 403 |
| 正常更新 | ✅ | 更新后数据正确 |
| 更新不存在的记录 | ✅ | 返回 400 或 404 |
| 正常删除 | ✅ | 删除后查询不到 |
| CSV/Excel 导出 | 建议 | 返回正确的文件类型 |
